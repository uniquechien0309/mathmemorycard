    <script src="https://unpkg.com/react@18/umd/react.production.min.js" crossorigin></script>

    <!-- 引入 Babel 用於瀏覽器內編譯 JSX -->
    <script src="https://unpkg.com/@babel/standalone/babel.min.js"></script>

    <style>
        /* 3D 翻牌動畫與基礎設定 */
        .perspective-1000 { perspective: 1000px; }
        .transform-style-3d { transform-style: preserve-3d; }

            background-size: 20px 20px;
        }

        /* 彈跳動畫 */
        @keyframes bounce-short {
            0%, 100% { transform: translateX(-50%) translateY(0); }

    <div id="root"></div>

    <script type="text/babel">
        const { useState, useEffect } = React;

        // --- SVG 圖標組件 ---
        const IconUser = () => (
            <svg xmlns="http://www.w3.org/2000/svg" width="20" height="20" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round"><path d="M19 21v-2a4 4 0 0 0-4-4H9a4 4 0 0 0-4 4v2"/><circle cx="12" cy="7" r="4"/></svg>
        );
        const IconTrophy = () => (
            <svg xmlns="http://www.w3.org/2000/svg" width="48" height="48" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round" className="text-yellow-500"><path d="M6 9H4.5a2.5 2.5 0 0 1 0-5H6"/><path d="M18 9h1.5a2.5 2.5 0 0 0 0-5H18"/><path d="M4 22h16"/><path d="M10 14.66V17c0 .55-.47.98-.97 1.21C7.85 18.75 7 20.24 7 22"/><path d="M14 14.66V17c0 .55.47.98.97 1.21C16.15 18.75 17 20.24 17 22"/><path d="M18 2H6v7a6 6 0 0 0 12 0V2Z"/></svg>
        );

        // --- 數學題目生成邏輯 ---
        function genProblem(target, type) {
            const isAdd = Math.random() > 0.5;
            const limit = type === '3digit' ? 999 : 99;
            if (isAdd) {
                const a = Math.floor(Math.random() * (target - 20)) + 10;
                return `${a} + ${target - a}`;
            } else {
                const a = Math.floor(Math.random() * (limit - (target + 10))) + (target + 10);
                return `${a} - ${a - target}`;
            }
        }

        function generateCards(count, type) {
            let cards = [];
            let usedValues = new Set();
            const pairCount = count / 2;

            for (let i = 0; i < pairCount; i++) {
                let target, eq1, eq2;
                
                if (type === 'multiplication') {
                    let a, b;
                    do {
                        a = Math.floor(Math.random() * 18) + 2; // 2~19 (一位數或十幾)
                        b = Math.floor(Math.random() * 8) + 2;  // 2~9 (一位數)
                        target = a * b;
                    } while (usedValues.has(target));
                    eq1 = `${a} × ${b}`;
                    eq2 = Math.random() > 0.5 ? `${b} × ${a}` : `${a} × ${b}`;
                } else {
                    const min = type === '3digit' ? 150 : 25;
                    const range = type === '3digit' ? 700 : 60;
                    do { target = Math.floor(Math.random() * range) + min; } while (usedValues.has(target));
                    eq1 = genProblem(target, type);
                    eq2 = genProblem(target, type);
                }
                usedValues.add(target);
                
                // 決定牌面：20% 都是答案, 40% 一題一答, 40% 都是題目
                const rand = Math.random();
                let c1, c2;
                if (rand < 0.2) { c1 = `${target}`; c2 = `${target}`; }
                else if (rand < 0.6) { c1 = eq1; c2 = `${target}`; }
                else { c1 = eq1; c2 = eq2; }

                cards.push({ id: `p${i}a`, content: c1, value: target, isMatched: false, isFlipped: false });
                cards.push({ id: `p${i}b`, content: c2, value: target, isMatched: false, isFlipped: false });
            }
            
            // 打亂卡牌
            return cards.sort(() => Math.random() - 0.5);
        }

        // --- 主應用組件 ---
        const App = () => {
            // 系統狀態
            const [gameState, setGameState] = useState('setup'); // 'setup', 'playing', 'gameover'
            
            // 設定狀態
            const [problemType, setProblemType] = useState('3digit');
            const [cardCount, setCardCount] = useState(16);
            const [playerCount, setPlayerCount] = useState(4);
            const [players, setPlayers] = useState([
                {name:'', score:0}, {name:'', score:0}, {name:'', score:0}, {name:'', score:0}
            ]);
            
            // 遊戲中狀態
            const [cards, setCards] = useState([]);
            const [currentPlayerIndex, setCurrentPlayerIndex] = useState(0);
            const [flippedIndices, setFlippedIndices] = useState([]);
            const [actionPending, setActionPending] = useState(false);
            const [feedback, setFeedback] = useState(null);
            const [startTime, setStartTime] = useState(null);
            const [endTime, setEndTime] = useState(null);
            const [showExitConfirm, setShowExitConfirm] = useState(false);

            // 開始遊戲
            const startGame = () => {
                // 初始化玩家分數，未填名字自動給預設名稱
                const finalPlayers = players.slice(0, playerCount).map((p, i) => ({
                    ...p, 
                    name: p.name.trim() === '' ? `玩家 ${i+1}` : p.name, 
                    score: 0
                }));
                
                setPlayers(finalPlayers);
                setCards(generateCards(cardCount, problemType));
                setCurrentPlayerIndex(0);
                setFlippedIndices([]);
                setActionPending(false);
                setFeedback(null);
                setStartTime(Date.now());
                setShowExitConfirm(false);
                setGameState('playing');
            };

            // 翻牌處理
            const handleCardClick = (i) => {
                // 如果有動作等待中、牌已翻開、牌已配對，就不處理
                if (actionPending || cards[i].isFlipped || cards[i].isMatched || flippedIndices.length >= 2) return;
                
                // 修正：不可直接竄改原狀態物件，必須建立全新的物件拷貝
                setCards(prev => prev.map((c, idx) => idx === i ? { ...c, isFlipped: true } : c));
                
                const newFlipped = [...flippedIndices, i];
                setFlippedIndices(newFlipped);
                
                if (newFlipped.length === 2) setActionPending(true);
            };

            // 回合結算邏輯
            const finalizeTurn = (isMatch) => {
                const currentFlipped = flippedIndices;
                setTimeout(() => {
                    setCards(prev => {
                        const next = prev.map((c, idx) => {
                            if (idx === currentFlipped[0] || idx === currentFlipped[1]) {
                                return isMatch ? { ...c, isMatched: true } : { ...c, isFlipped: false };
                            }
                            return c;
                        });
                        
                        // 檢查遊戲是否結束
                        if (next.every(c => c.isMatched)) { 
                            setEndTime(Date.now()); 
                            setGameState('gameover'); 
                        }
                        return next;
                    });
                    
                    setFlippedIndices([]);
                    setActionPending(false);
                    setFeedback(null);
                    setCurrentPlayerIndex(p => (p + 1) % playerCount); // 換下一位玩家
                }, 1500); // 顯示提示 1.5 秒後換人
            };

            // 點擊「碰」
            const handlePong = () => {
                const match = cards[flippedIndices[0]].value === cards[flippedIndices[1]].value;
                if (match) {
                    setFeedback({ text: "答對了！得 2 分", type: 'success' });
                    // 修正：以安全的方式更新分數
                    setPlayers(prev => prev.map((p, idx) => idx === currentPlayerIndex ? { ...p, score: p.score + 2 } : p));
                    finalizeTurn(true);
                } else {
                    setFeedback({ text: "答案不同喔！算錯囉", type: 'error' });
                    finalizeTurn(false);
                }
            };

            // 點擊「放回」
            const handlePass = () => {
                const match = cards[flippedIndices[0]].value === cards[flippedIndices[1]].value;
                setFeedback({ 
                    text: match ? "哎呀，其實答案是一樣的喔！" : "判斷正確！安全換下一位", 
                    type: match ? 'error' : 'info' 
                });
                finalizeTurn(false);
            };

            // ================= 渲染畫面 =================

            // 1. 設定畫面
            if (gameState === 'setup') return (
                <div className="max-w-md mx-auto my-10 bg-white p-8 rounded-3xl shadow-xl border border-slate-100">
                    <h1 className="text-3xl md:text-4xl font-black text-center text-blue-600 mb-8">數學翻翻樂</h1>
                    
                    {/* 題型選擇 (橫向排版) */}
                    <div className="mb-6">
                        <label className="block font-bold mb-2 text-slate-700">選擇題型</label>
                        <div className="flex gap-2">
                            {[
                                {id: '2digit', label: '兩位數加減'},
                                {id: '3digit', label: '三位數加減'},
                                {id: 'multiplication', label: '乘法'}
                            ].map(t => (
                                <button key={t.id} onClick={()=>setProblemType(t.id)} 
                                    className={`flex-1 py-3 rounded-xl font-bold text-sm md:text-base transition-all ${problemType===t.id ? 'bg-purple-600 text-white shadow-md scale-105' : 'bg-slate-100 text-slate-600 hover:bg-slate-200'}`}>
                                    {t.label}
                                </button>
                            ))}
                        </div>
                    </div>

                    {/* 人數與難度 */}
                    <div className="mb-6">
                        <label className="block font-bold mb-2 text-slate-700">選擇玩家人數</label>
                        <div className="flex gap-2 mb-4">
                            {[1,2,3,4].map(n => (
                                <button key={n} onClick={()=>setPlayerCount(n)} 
                                    className={`flex-1 py-2 rounded-xl font-bold transition-all ${playerCount===n ? 'bg-green-600 text-white shadow-md' : 'bg-slate-100 text-slate-600 hover:bg-slate-200'}`}>
                                    {n} 人
                                </button>
                            ))}
                        </div>
                        
                        <label className="block font-bold mb-2 text-slate-700">選擇牌卡數量 (難度)</label>
                        <div className="flex gap-2">
                            {[8,16,24].map(n => (
                                <button key={n} onClick={()=>setCardCount(n)} 
                                    className={`flex-1 py-2 rounded-xl font-bold transition-all ${cardCount===n ? 'bg-blue-600 text-white shadow-md' : 'bg-slate-100 text-slate-600 hover:bg-slate-200'}`}>
                                    {n} 張
                                </button>
                            ))}
                        </div>
                    </div>

                    {/* 玩家名稱設定 */}
                    <div className="space-y-3 mb-8">
                        <label className="block font-bold text-slate-700">玩家名稱設定</label>
                        {Array.from({length: playerCount}).map((_, i) => (
                            <input key={i} type="text" placeholder={`玩家 ${i+1}`} value={players[i]?.name || ''} 
                                onChange={e => {
                                    // 修正：玩家名稱的安全更新
                                    const val = e.target.value; 
                                    setPlayers(prev => prev.map((p, idx) => idx === i ? { ...p, name: val } : p));
                                }}
                                className="w-full px-4 py-2 border-2 border-slate-200 rounded-xl focus:border-blue-400 outline-none" 
                            />
                        ))}
                    </div>

                    <button onClick={startGame} className="w-full py-4 bg-gradient-to-r from-blue-500 to-indigo-600 text-white rounded-2xl font-black text-xl shadow-lg hover:scale-[1.02] transition-all">
                        開始挑戰！
                    </button>
                </div>
            );

            // 2. 遊戲進行畫面
            if (gameState === 'playing') return (
                <div className="max-w-6xl mx-auto px-4 pt-16 pb-64 min-h-screen relative">
                    {/* 返回按鈕 */}
                    <button onClick={() => setShowExitConfirm(true)} 
                        className="absolute top-4 right-4 bg-white/90 backdrop-blur flex items-center gap-1 px-4 py-2 rounded-xl shadow-md font-bold text-slate-600 border border-slate-200 z-10 hover:bg-slate-50 transition-colors">
                        返回設定
                    </button>
                    
                    {/* 頂部計分板 */}
                    <div className="flex flex-wrap gap-4 mb-8 justify-center">
                        {players.slice(0, playerCount).map((p, i) => (
                            <div key={i} className={`flex-1 min-w-[120px] max-w-[200px] p-4 rounded-2xl border-2 transition-all duration-300 ${i===currentPlayerIndex ? 'ring-4 ring-blue-400 bg-white shadow-xl scale-110 z-10' : 'opacity-70 bg-slate-50 border-slate-200'}`}>
                                <div className={`font-bold flex items-center gap-1 mb-1 ${i===currentPlayerIndex?'text-blue-600':'text-slate-600'}`}>
                                    <IconUser /> {p.name}
                                </div>
                                <div className="text-3xl font-black">{p.score} <span className="text-base font-normal text-slate-500">分</span></div>
                            </div>
                        ))}
                    </div>

                    {/* 卡牌網格區 */}
                    <div className={`grid gap-3 md:gap-4 max-w-5xl mx-auto w-full ${cardCount===8 ? 'grid-cols-2 md:grid-cols-4' : cardCount===16 ? 'grid-cols-4' : 'grid-cols-4 md:grid-cols-6'}`}>
                        {cards.map((c, i) => (
                            <div key={c.id} onClick={()=>handleCardClick(i)} className={`aspect-[3/4] perspective-1000 cursor-pointer ${c.isMatched ? 'opacity-0 pointer-events-none transition-all duration-1000' : ''}`}>
                                <div className={`relative w-full h-full transform-style-3d card-transition ${c.isFlipped ? 'rotate-y-180' : ''}`}>
                                    {/* 背面 */}
                                    <div className="absolute inset-0 backface-hidden card-pattern rounded-2xl border-4 border-white shadow-md flex items-center justify-center">
                                        <div className="w-12 h-12 bg-white/20 rounded-full flex items-center justify-center text-white font-black text-3xl">?</div>
                                    </div>
                                    {/* 正面 */}
                                    <div className="absolute inset-0 backface-hidden rotate-y-180 bg-white rounded-2xl border-4 border-slate-200 shadow-lg flex items-center justify-center p-2 text-center">
                                        <span className={`font-black tracking-wider ${c.content.includes('+') || c.content.includes('-') || c.content.includes('×') ? 'text-2xl md:text-3xl lg:text-4xl text-slate-800' : 'text-4xl md:text-5xl lg:text-6xl text-blue-600'}`}>
                                            {c.content}
                                        </span>
                                    </div>
                                </div>
                            </div>
                        ))}
                    </div>

                    {/* 碰/放回 選項彈窗 (半透明、置底) */}
                    {actionPending && !feedback && (
                        <div className="fixed bottom-8 left-1/2 transform -translate-x-1/2 w-[95%] max-w-lg bg-white/90 backdrop-blur-lg p-5 rounded-3xl shadow-2xl border border-slate-200 z-50 animate-bounce-short">
                            <p className="text-center font-bold text-slate-700 text-lg mb-4">你覺得這兩張的答案一樣嗎？</p>
                            <div className="flex gap-4">
                                <button onClick={handlePong} className="flex-1 bg-red-500 hover:bg-red-600 text-white py-4 rounded-2xl font-black text-2xl shadow-lg transition-transform hover:scale-105">
                                    相同！碰！
                                </button>
                                <button onClick={handlePass} className="flex-1 bg-slate-200 hover:bg-slate-300 text-slate-700 py-4 rounded-2xl font-bold text-xl shadow-md transition-transform hover:scale-105">
                                    不同，放回
                                </button>
                            </div>
                        </div>
                    )}

                    {/* 正確/錯誤 回饋提示 */}
                    {feedback && (
                        <div className="fixed inset-0 flex items-center justify-center z-[60] pointer-events-none">
                            <div className={`px-8 py-6 rounded-3xl text-white font-black text-2xl md:text-3xl shadow-2xl transform scale-110 ${feedback.type==='success' ? 'bg-green-500' : feedback.type==='error' ? 'bg-red-500' : 'bg-blue-500'}`}>
                                {feedback.text}
                            </div>
                        </div>
                    )}

                    {/* 退出確認視窗 */}
                    {showExitConfirm && (
                        <div className="fixed inset-0 flex items-center justify-center z-[100]">
                            <div className="absolute inset-0 bg-black/50 backdrop-blur-sm" onClick={() => setShowExitConfirm(false)}></div>
                            <div className="relative bg-white p-8 rounded-3xl shadow-2xl text-center max-w-sm w-11/12 animate-bounce-short">
                                <h3 className="text-2xl font-black mb-2 text-slate-800">確定要結束嗎？</h3>
                                <p className="text-slate-500 mb-6 font-bold">遊戲進度將會遺失喔！</p>
                                <div className="flex gap-3">
                                    <button onClick={()=>{setGameState('setup'); setShowExitConfirm(false);}} className="flex-1 bg-red-500 text-white py-3 rounded-xl font-bold text-lg hover:bg-red-600 transition-colors">
                                        確定退出
                                    </button>
                                    <button onClick={()=>setShowExitConfirm(false)} className="flex-1 bg-slate-200 text-slate-700 py-3 rounded-xl font-bold text-lg hover:bg-slate-300 transition-colors">
                                        繼續遊戲
                                    </button>
                                </div>
                            </div>
                        </div>
                    )}
                </div>
            );

            // 3. 遊戲結束畫面
            if (gameState === 'gameover') {
                const activePlayers = players.slice(0, playerCount);
                const maxScore = Math.max(...activePlayers.map(p => p.score));
                const winners = activePlayers.filter(p => p.score === maxScore);
                const timeSec = Math.floor((endTime - startTime) / 1000);
                
                return (
                    <div className="flex items-center justify-center min-h-screen p-4 bg-slate-800/30 backdrop-blur-sm">
                        <div className="bg-white p-10 rounded-[3rem] shadow-2xl text-center max-w-md w-full border-8 border-blue-50">
                            <div className="flex justify-center mb-4"><IconTrophy /></div>
                            <h2 className="text-4xl font-black mb-4 text-slate-800">遊戲完成！</h2>
                            
                            <div className="text-2xl font-bold text-yellow-600 mb-2">
                                🏆 {winners.map(w => w.name).join(' & ')}
                            </div>
                            
                            <div className="text-blue-600 font-bold bg-blue-50 py-2 rounded-full mb-6 inline-block px-6 border border-blue-100">
                                ⏱️ 完成時間：{Math.floor(timeSec / 60) > 0 ? `${Math.floor(timeSec / 60)}分 ` : ''}{timeSec % 60}秒
                            </div>
                            
                            <div className="space-y-3 mb-8 text-left">
                                {activePlayers.map((p, i) => (
                                    <div key={i} className={`flex justify-between items-center bg-slate-50 p-4 rounded-xl font-bold border ${p.score === maxScore ? 'border-yellow-400 bg-yellow-50' : 'border-slate-100'}`}>
                                        <span className="text-slate-700">{p.name}</span>
                                        <span className="text-blue-600 text-xl">{p.score} 分</span>
                                    </div>
                                ))}
                            </div>
                            
                            <button onClick={()=>setGameState('setup')} className="w-full py-4 bg-blue-600 hover:bg-blue-700 transition-colors text-white rounded-2xl font-black text-xl shadow-lg hover:scale-105">
                                返回設定重新開始
                            </button>
                        </div>
                    </div>
                );
            }
        };

        const root = ReactDOM.createRoot(document.getElementById('root'));
        root.render(<App />);
    </script>
</body>
